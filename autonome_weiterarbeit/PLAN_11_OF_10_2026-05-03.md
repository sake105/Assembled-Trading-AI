```python
"""Pairs-trading strategy — mean-reversion on cointegrated pairs.

Wrapper around signals/pairs_trading.py for ensemble integration.
"""
from src.assembled_core.signals.pairs_trading import (
    cointegration_score, kalman_hedge_ratio, spread_z_score, generate_pairs_signals,
)

class PairsTradingStrategy:
    name = "pairs_trading_v1"
    
    def __init__(self, config: dict):
        self.lookback = config.get("lookback_days", 252)
        self.min_cointegration = config.get("min_cointegration_p", 0.05)
        self.entry_z = config.get("entry_zscore", 2.0)
        self.exit_z = config.get("exit_zscore", 0.5)
    
    def discover_pairs(self, prices: pd.DataFrame) -> list[tuple[str, str]]:
        """Quarterly: rediscover cointegrated pairs."""
        symbols = prices["symbol"].unique()
        pairs = []
        for i, s1 in enumerate(symbols):
            for s2 in symbols[i+1:]:
                p_value = cointegration_score(
                    prices[prices.symbol==s1]["close"],
                    prices[prices.symbol==s2]["close"],
                )
                if p_value < self.min_cointegration:
                    pairs.append((s1, s2))
        return pairs[:20]  # top 20 pairs
    
    def generate_signals(self, prices: pd.DataFrame, pairs: list) -> pd.DataFrame:
        return generate_pairs_signals(prices, pairs, 
                                       entry_z=self.entry_z, exit_z=self.exit_z)
```

## 2.2 Monte-Carlo-Trade-Resampling (3-4h)

Aktuell: ein Backtest = ein Pfad. Das ist **eine Realisierung** aus einer ganzen Verteilung. Monte-Carlo permutiert die **Reihenfolge** deiner 233 OOS-Trades — wenn die Trade-Reihenfolge zufällig wäre, wie sähen Sharpe/MDD-Verteilungen aus?

### Konkrete Schritte

**Schritt 2.2.1 — `qa/monte_carlo_paths.py` (neu, ~150 LOC, 2h)**

```python
"""Monte-Carlo trade-order simulation for backtest robustness.

Permutes the order of historical trades to estimate the distribution of
equity-curve metrics under random trade sequencing. This separates
'edge' (mean trade return > 0) from 'lucky sequencing' (specific order
of wins/losses produced low MDD).
"""
from __future__ import annotations
import numpy as np
import pandas as pd

def monte_carlo_trade_paths(
    trades: pd.DataFrame,  # columns: pnl, return_pct, holding_days
    initial_capital: float = 100_000,
    n_paths: int = 5000,
    seed: int = 42,
) -> dict:
    rng = np.random.default_rng(seed)
    
    pnl_array = trades["pnl"].values
    n_trades = len(pnl_array)
    
    sharpe_list = []
    mdd_list = []
    cagr_list = []
    final_equity_list = []
    
    for _ in range(n_paths):
        # Permute trade order
        permuted = rng.permutation(pnl_array)
        cum_pnl = np.cumsum(permuted)
        equity = initial_capital + cum_pnl
        
        # Compute metrics
        peak = np.maximum.accumulate(equity)
        drawdown = (equity - peak) / peak
        mdd_list.append(drawdown.min())
        
        returns = np.diff(equity) / equity[:-1]
        sharpe = returns.mean() / returns.std() * np.sqrt(252) if returns.std() > 0 else 0
        sharpe_list.append(sharpe)
        
        years = n_trades / 252  # approximation
        cagr = (equity[-1] / initial_capital) ** (1 / years) - 1
        cagr_list.append(cagr)
        final_equity_list.append(equity[-1])
    
    return {
        "n_paths": n_paths,
        "sharpe": {
            "mean": float(np.mean(sharpe_list)),
            "std": float(np.std(sharpe_list)),
            "p10": float(np.percentile(sharpe_list, 10)),
            "p50": float(np.percentile(sharpe_list, 50)),
            "p90": float(np.percentile(sharpe_list, 90)),
        },
        "mdd": {
            "mean": float(np.mean(mdd_list)),
            "p10": float(np.percentile(mdd_list, 10)),
            "p50": float(np.percentile(mdd_list, 50)),
            "p90": float(np.percentile(mdd_list, 90)),
            "p99": float(np.percentile(mdd_list, 99)),  # worst case
        },
        "cagr": {
            "mean": float(np.mean(cagr_list)),
            "p10": float(np.percentile(cagr_list, 10)),
            "p50": float(np.percentile(cagr_list, 50)),
            "p90": float(np.percentile(cagr_list, 90)),
        },
    }
```

```python
"""Stress-test backtest across historical crises.

Runs the same strategy across N defined stress windows.
Outputs per-crisis CAGR, MDD, Sharpe, and worst-day return.
Aggregates to "stress score" — geometric mean of per-window CAGR.
"""

import yaml
import json
import subprocess
from pathlib import Path

with open("configs/stress_windows.yaml") as f:
    stress_cfg = yaml.safe_load(f)

results = []
for window in stress_cfg["stress_windows"]:
    print(f"Stress test: {window['name']} ({window['start']} → {window['end']})")
    
    output_path = Path(f"output/stress/{window['name']}.json")
    
    subprocess.run([
        "python", "scripts/run_backtest_strategy.py",
        "--start", window["start"],
        "--end", window["end"],
        "--out", str(output_path),
        "--policy", "configs/policy.yaml",
    ], check=True)
    
    with open(output_path) as f:
        bt = json.load(f)
    
    results.append({
        "window": window["name"],
        "description": window["description"],
        "cagr": bt["metrics"]["cagr"],
        "sharpe": bt["metrics"]["sharpe"],
        "mdd": bt["metrics"]["mdd"],
        "worst_day": bt["metrics"].get("worst_day_return", 0),
        "n_trades": bt["metrics"]["n_trades"],
    })

# Aggregate stress score
import numpy as np
stress_score = np.exp(np.mean([np.log(1 + r["cagr"]) for r in results])) - 1
worst_mdd = min(r["mdd"] for r in results)
worst_day = min(r["worst_day"] for r in results)

stress_summary = {
    "windows": results,
    "aggregate": {
        "stress_score_cagr": stress_score,  # geometric mean
        "worst_mdd_across_crises": worst_mdd,
        "worst_single_day": worst_day,
    },
}

with open("output/stress/aggregate.json", "w") as f:
    json.dump(stress_summary, f, indent=2)

print(f"\n=== STRESS TEST AGGREGATE ===")
print(f"Stress-Score CAGR (geom mean): {stress_score:.2%}")
print(f"Worst MDD across crises:       {worst_mdd:.2%}")
print(f"Worst single day:              {worst_day:.2%}")
```

```python
async def paper_live_loop(symbols: list[str], cycle_minutes: int = 15):
    """Run trading cycle every N minutes during market hours."""
    
    if is_kill_switch_engaged():
        logger.critical("Kill-switch is engaged. Cannot start.")
        return
    
    broker = AlpacaAdapter.from_env()  # reads ALPACA_API_KEY/SECRET
    
    while True:
        now = datetime.now(timezone.utc)
        
        # Market hours check (9:30 ET to 16:00 ET = 13:30 to 20:00 UTC, weekdays)
        if not _is_market_hours(now):
            logger.info(f"[live] outside market hours, sleeping 5min")
            await asyncio.sleep(300)
            continue
        
        if is_kill_switch_engaged():
            logger.critical("Kill-switch tripped during loop. Exiting.")
            break
        
        try:
            # 1. Pull latest data
            ctx = build_context(
                symbols=symbols,
                as_of=now,
                mode="paper",
                broker=broker,
            )
            
            # 2. Run cycle
            result = run_trading_cycle(ctx)
            
            # 3. Submit orders
            if result.orders_filtered is not None and not result.orders_filtered.empty:
                exec_result = execute_orders_via_broker(
                    broker=broker, orders=result.orders_filtered
                )
                logger.info(f"[live] {len(exec_result.submitted)} orders submitted, "
                           f"{len(exec_result.filled)} fills")
            else:
                logger.info(f"[live] no orders this cycle")
            
            # 4. Daily summary write
            _write_paper_live_summary(result, exec_result)
            
        except Exception as e:
            logger.error(f"[live] cycle error: {e}", exc_info=True)
            # Don't kill on cycle errors — log and continue
        
        # Wait until next cycle
        await asyncio.sleep(cycle_minutes * 60)


def _is_market_hours(now: datetime) -> bool:
    """Check if currently within US equity market hours."""
    if now.weekday() >= 5:  # Sat/Sun
        return False
    # Convert to ET
    import pytz
    et_now = now.astimezone(pytz.timezone("America/New_York"))
    return (et_now.hour, et_now.minute) >= (9, 30) and \
           (et_now.hour, et_now.minute) < (16, 0)


def _write_paper_live_summary(result, exec_result):
    """Write per-cycle summary for daily aggregation."""
    from pathlib import Path
    import json
    
    out_dir = Path("output/paper_live") / datetime.now().strftime("%Y-%m-%d")
    out_dir.mkdir(parents=True, exist_ok=True)
    
    cycle_id = datetime.now().strftime("%H%M%S")
    summary = {
        "cycle_id": cycle_id,
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "n_signals": len(result.signals) if result.signals is not None else 0,
        "n_orders_submitted": len(exec_result.submitted),
        "n_orders_filled": len(exec_result.filled),
        "exposure_multiplier": result.meta.get("final_multiplier"),
        "regime": result.meta.get("regime"),
    }
    with open(out_dir / f"cycle_{cycle_id}.json", "w") as f:
        json.dump(summary, f, indent=2)
```

```python
"""Quantify the sim-to-real gap between backtest and paper-live.

Compares:
- Backtest fills (modeled) vs paper-live fills (real broker)
- Slippage attribution: spread, queue position, latency, market impact
- Daily P&L variance breakdown
"""

def analyze_sim_to_real_gap(
    backtest_period_results: dict,
    paper_live_period_results: dict,
) -> dict:
    """Compare backtest vs paper-live for the same period."""
    
    bt_metrics = backtest_period_results["metrics"]
    pl_metrics = paper_live_period_results["metrics"]
    
    sharpe_gap = pl_metrics["sharpe"] - bt_metrics["sharpe"]
    cagr_gap = pl_metrics["cagr"] - bt_metrics["cagr"]
    
    # Decompose slippage
    bt_avg_slip = backtest_period_results.get("avg_slippage_bps", 0)
    pl_avg_slip = paper_live_period_results.get("avg_slippage_bps", 0)
    slip_gap_bps = pl_avg_slip - bt_avg_slip
    
    return {
        "period": {
            "start": pl_metrics.get("start"),
            "end": pl_metrics.get("end"),
            "n_trading_days": pl_metrics.get("n_trading_days"),
        },
        "backtest": bt_metrics,
        "paper_live": pl_metrics,
        "gap": {
            "sharpe_drop": sharpe_gap,
            "cagr_drop": cagr_gap,
            "slippage_excess_bps": slip_gap_bps,
        },
        "verdict": _classify_gap(sharpe_gap, slip_gap_bps),
    }


def _classify_gap(sharpe_drop, slip_gap_bps):
    """Classify the sim-to-real gap."""
    if sharpe_drop > -0.3 and slip_gap_bps < 5:
        return "EXCELLENT — backtest realistic, ready to scale"
    elif sharpe_drop > -0.7 and slip_gap_bps < 15:
        return "ACCEPTABLE — typical sim-to-real gap, ready for live with caution"
    elif sharpe_drop > -1.5:
        return "CONCERNING — significant gap, investigate before live"
    else:
        return "ALARMING — backtest unrealistic, do not go live"
```

```python
"""Multi-channel alerting system.

Reads configs/alerting.yaml. Subscribers register handlers; events
are dispatched to all matching channels with cooldown logic.
"""

import yaml
import time
from pathlib import Path
from collections import defaultdict

class AlertManager:
    def __init__(self, config_path: str = "configs/alerting.yaml"):
        with open(config_path) as f:
            self.config = yaml.safe_load(f)
        self.last_fired: dict[str, float] = defaultdict(float)
    
    def fire(self, rule_name: str, context: dict):
        """Fire an alert if rule matches and cooldown has passed."""
        rule = self._find_rule(rule_name)
        if not rule:
            return
        
        # Cooldown check
        cooldown_min = rule.get("cooldown_minutes", 30)
        last = self.last_fired[rule_name]
        if cooldown_min > 0 and time.time() - last < cooldown_min * 60:
            return  # still in cooldown
        
        # Format message
        message = rule["message"].format(**context)
        
        # Dispatch to all matching channels
        severity = rule["severity"]
        for channel_cfg in self.config["channels"].get(severity, []):
            self._dispatch(channel_cfg, message, context)
        
        self.last_fired[rule_name] = time.time()
    
    def _dispatch(self, channel_cfg, message, context):
        kind = channel_cfg["type"]
        if kind == "telegram":
            self._send_telegram(channel_cfg, message)
        elif kind == "email":
            self._send_email(channel_cfg, message)
        # ... etc
```

## 5.2 Pre-Trade-Sanity-Checks (3-4h)

Bevor ein Trade rausgeht: prüfe ob er "Sinn macht". Das ist nicht Risk-Gate (rejection-Kriterium), sondern **flagging-Kriterium** mit Halt-Option.

### Konkrete Schritte

**Schritt 5.2.1 — `qa/sanity_checks.py` (neu, ~150 LOC, 2h)**

```python
"""Pre-trade sanity checks — catch obviously wrong trades before they go out.

These are NOT hard risk gates (those reject). These are ANOMALY detectors
that flag for review and optionally hold the trade.

Examples of catches:
- Position 5x larger than typical for this symbol
- Trade direction opposite to last 3 trades on same symbol (whipsaw?)
- Buy in stock down 8% today on news (data lag?)
- Order at price 2% from current quote (stale quote?)
- Unusual symbol (not in PIT-universe?)
"""

class SanityChecker:
    def __init__(self, history_dir: Path):
        self.history_dir = history_dir
        self.recent_trades = self._load_recent_trades(days=30)
    
    def check_order(self, order: dict, market_state: dict) -> dict:
        """Run all sanity checks. Returns dict with flags + halt_recommendation."""
        flags = []
        halt = False
        
        # Check 1: Position size vs historical
        sym = order["symbol"]
        sym_recent = self.recent_trades[self.recent_trades.symbol == sym]
        if not sym_recent.empty:
            avg_size = sym_recent["qty"].abs().mean()
            if order["qty"] > avg_size * 5:
                flags.append({
                    "rule": "position_5x_typical",
                    "severity": "high",
                    "detail": f"qty {order['qty']} vs avg {avg_size:.0f}",
                })
                halt = True
        
        # Check 2: Whipsaw detection
        last_3 = sym_recent.tail(3)
        if len(last_3) == 3 and (last_3["side"] != order["side"]).all():
            flags.append({
                "rule": "whipsaw_3x_reverse",
                "severity": "medium",
                "detail": "All last 3 trades on this symbol opposite direction",
            })
        
        # Check 3: Price/quote staleness
        if "last_quote" in market_state:
            quote_age_sec = (pd.Timestamp.now(tz="UTC") - market_state["last_quote_time"]).total_seconds()
            if quote_age_sec > 60:
                flags.append({
                    "rule": "quote_stale",
                    "severity": "medium",
                    "detail": f"Last quote was {quote_age_sec:.0f}s ago",
                })
            
            quote_drift_pct = abs(order["limit_price"] - market_state["last_quote"]) / market_state["last_quote"]
            if quote_drift_pct > 0.02:
                flags.append({
                    "rule": "limit_far_from_quote",
                    "severity": "high",
                    "detail": f"Limit {order['limit_price']:.2f} vs quote {market_state['last_quote']:.2f} ({quote_drift_pct:.1%} away)",
                })
                halt = True
        
        # Check 4: Buying despite negative news today
        if order["side"] == "buy" and market_state.get("daily_news_sentiment", 0) < -0.5:
            flags.append({
                "rule": "buy_against_news",
                "severity": "low",
                "detail": f"Daily sentiment: {market_state['daily_news_sentiment']:.2f}",
            })
        
        # Check 5: Universe membership (PIT)
        if not market_state.get("symbol_in_pit_universe", True):
            flags.append({
                "rule": "symbol_not_in_pit_universe",
                "severity": "critical",
                "detail": f"{sym} not in PIT-universe for {order.get('as_of')}",
            })
            halt = True
        
        return {
            "n_flags": len(flags),
            "flags": flags,
            "halt_recommendation": halt,
            "max_severity": max((f["severity"] for f in flags), default="none"),
        }
```

```python
            for flag in check_result["flags"]:
                log.warning(f"  - [{flag['severity']}] {flag['rule']}: {flag['detail']}")
        
        # Halt if recommended (configurable)
        sanity_cfg = (policy.get("sanity_checks") or {})
        if check_result["halt_recommendation"] and sanity_cfg.get("halt_on_high_severity", True):
            halted_orders.append((order, check_result))
            
            # Alert
            from src.assembled_core.ops.alerting import AlertManager
            AlertManager().fire("sanity_check_halt", {
                "symbol": order["symbol"],
                "flags": check_result["flags"],
            })
    
    # Filter halted orders out
    if halted_orders:
        halted_idx = [h[0].name for h in halted_orders]
        result.orders_filtered = result.orders_filtered.drop(index=halted_idx)
        log.warning(f"[sanity] {len(halted_orders)} orders halted by sanity check")
```

```python
python scripts/release_sanity_halt.py --order-id <id> --reason "Legitimate scaled entry, confirmed manually"
```
